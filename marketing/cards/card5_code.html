async def _process_message(raw: dict):
    """解析 → 幂等 → 线程池执行 → 失败闭环"""
    try:
        event = parse_event(raw)      # 坏消息直接死信
    except EventParseError as e:
        await _dead_letter_raw(raw, str(e))
        return

    # 幂等：已成功消费的 event_id 直接跳过
    if dedup.is_processed(event.event_id):
        metrics.inc_counter("kafka_events_total",
            {"outcome": "duplicate_skipped"})
        return

    try:
        await loop.run_in_executor(None,
                                   _execute_with_retry, event)
    except Exception as e:
        # 重试耗尽 → 死信 + 元数据标记 failed
        await _dead_letter_raw(event.model_dump(), str(e))
        meta_repo.mark_failed(event.file_title, str(e))
        return

    dedup.mark_processed(event.event_id)  # 消费成功
